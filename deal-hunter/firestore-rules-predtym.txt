rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ═══════════════════════════════════════════════════════════════
    //  KRITICKÁ OPRAVA – nie je to kozmetika, prečítaj si to
    //
    //  Pôvodne tu bolo len overenie, že je užívateľ prihlásený heslom,
    //  a obmedzenie na konkrétny e-mail bolo zakomentované.
    //
    //  Lenže registrácia e-mailom a heslom je v tomto projekte otvorená.
    //  Ktokoľvek si cez verejné Firebase API vytvorí účet za sekundu.
    //  Overené útokom: cudzí účet takto získal plné práva admina –
    //  čítanie čakajúcich dealov, schvaľovanie, úpravu aj mazanie
    //  čohokoľvek na stránke.
    //
    //  Preto je e-mail odteraz povinný. Bez neho je admin panel
    //  otvorený celému internetu.
    // ═══════════════════════════════════════════════════════════════
    //  Od 1. 10. 2026 aj overený e-mail: registrácia návštevníkov je
    //  otvorená a e-mail bez overenia by inak stačilo len napísať.
    function overenyUcet() {
      return request.auth != null
             && request.auth.token.firebase.sign_in_provider == 'password'
             && request.auth.token.email_verified == true;
    }
    // Majiteľ je napevno - nikto mu práva nevezme ani omylom zmazaním
    // záznamu v kolekcii admini.
    function jeMajitel() {
      return overenyUcet() && request.auth.token.email == 'foresttt11@outlook.com';
    }
    // Ďalší ľudia s prístupom: admini/{email} s poľom rola. Zapisovať
    // do tejto kolekcie smie len majiteľ. get() sa volá len pre
    // prihlásených s overeným e-mailom - bežný návštevník tak žiadne
    // čítanie navyše nespôsobí.
    function rola() {
      return get(/databases/$(database)/documents/admini/$(request.auth.token.email)).data.rola;
    }
    function maRolu() {
      return overenyUcet() && exists(/databases/$(database)/documents/admini/$(request.auth.token.email));
    }
    function isAdmin() {
      return jeMajitel() || (maRolu() && rola() == 'admin');
    }
    // Moderátor schvaľuje a upravuje dealy, kódy a komentáre. Nastavenia,
    // používateľov, návštevnosť ani financie nevidí.
    function isModerator() {
      return isAdmin() || (maRolu() && rola() == 'moderator');
    }

    // Dokumenty od návštevníkov musia mať automatické ID (20 znakov,
    // písmená a číslice - tak ho tvorí addDoc). Vlastné ID s apostrofom
    // by sa v admine dostalo do onclick a spustilo kód. Pole "id" v
    // dátach by zas prekrylo skutočné ID dokumentu.
    // Je moderácia vypnutá? Keď áno, nový príspevok smie ísť rovno von -
    // presne to robí prepínač v admine. Čítame to tu, lebo status si
    // inak píše sám klient a dalo by sa to obísť mimo formulára.
    function bezModeracie(pole) {
      return exists(/databases/$(database)/documents/settings/moderation)
             && get(/databases/$(database)/documents/settings/moderation).data.get(pole, true) == false;
    }
    function autoId(id) {
      return id.matches('^[A-Za-z0-9]{20}$');
    }
    function bezCudzichId() {
      return !('id' in request.resource.data) && !('dealId' in request.resource.data);
    }

    match /deals/{dealId} {
      // Verejnosť vidí len schválené dealy. Admin vidí všetko.
      allow read: if resource.data.status == 'approved' || isModerator();

      // Admin môže vytvoriť čokoľvek (obnova zo zálohy, pridanie z odkazu).
      allow create: if isAdmin();
      // Nový deal od návštevníka. Zverejniť ho rovno ("approved") smie len
      // dôveryhodný prispievateľ - koho tam plánovač zapíše, určujú
      // pravidlá v admine. Inak deal čaká na schválenie.
      //
      // Doteraz tu stačilo, že si status napíšeš sám: kto obišiel
      // formulár, vedel si deal zverejniť bez moderácie. Toto to zatvára.
      //
      // authorUid je nepovinné, ale keď tam je, musí to byť naozaj tvoje
      // konto - inak by si si cudzím uid pripisoval body.
      allow create: if (request.resource.data.status == 'pending'
                        || (request.resource.data.status == 'approved'
                            && (bezModeracie('dealsRequired')
                                || (request.auth != null
                                    && exists(/databases/$(database)/documents/duveryhodni/$(request.auth.uid))))))
                    && (!('authorUid' in request.resource.data)
                        || (request.auth != null && request.resource.data.authorUid == request.auth.uid))
                    && bezCudzichId() && autoId(dealId)
                    && request.resource.data.author is string
                    && request.resource.data.author.size() > 0
                    && request.resource.data.author.size() < 40
                    && request.resource.data.title is string
                    && request.resource.data.title.size() > 0
                    && request.resource.data.title.size() < 200
                    && request.resource.data.store is string
                    && request.resource.data.dealPrice is number
                    && request.resource.data.dealPrice >= 0
                    && request.resource.data.url is string
                    // Odkaz musí byť webová adresa (nie javascript: a pod.)
                    // a dĺžky textov majú strop - rovnako ako formulár.
                    && request.resource.data.url.matches('^https?://[^ ]+$')
                    && request.resource.data.url.size() <= 1500
                    && request.resource.data.store.size() <= 60
                    && (!('description' in request.resource.data)
                        || (request.resource.data.description is string && request.resource.data.description.size() <= 2000))
                    && request.resource.data.votes == 0
                    && request.resource.data.comments == 0;

      // Hlasovanie: ktokoľvek prihlásený (aj anonymne) môže meniť len
      // "votes", a to najviac o 2 (prepnutie hlasu z + na - je delta*2).
      // Počítadlo komentárov sa smie zvýšiť o presne 1.
      // Admin môže upravovať čokoľvek.
      allow update: if isModerator()
                    || (request.auth != null
                        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['votes'])
                        && request.resource.data.votes >= resource.data.votes - 2
                        && request.resource.data.votes <= resource.data.votes + 2)
                    || (request.auth != null
                        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['comments'])
                        && request.resource.data.comments == resource.data.comments + 1);

      allow delete: if isAdmin();

      // Kto už hlasoval a ako. Návštevník smie čítať len svoj vlastný
      // záznam – transakcia pri hlasovaní si ho musí prečítať.
      match /voters/{voterId} {
        allow read: if isAdmin() || (request.auth != null && request.auth.uid == voterId);
        allow write: if request.auth != null && request.auth.uid == voterId;
      }
    }

    // Komentáre – top-level wildcard, aby fungovali collectionGroup
    // dotazy použité v admin paneli.
    match /{path=**}/comments/{commentId} {
      allow get: if resource.data.status == 'approved' || isModerator();
      allow list: if resource.data.status == 'approved' || isModerator();

      allow create: if isAdmin();

      allow create: if (request.resource.data.status == 'pending'
                        || (request.resource.data.status == 'approved' && bezModeracie('commentsRequired')))
                    && bezCudzichId() && autoId(commentId)
                    && request.resource.data.nickname is string
                    && request.resource.data.nickname.size() > 0
                    && request.resource.data.nickname.size() < 40
                    && request.resource.data.text is string
                    && request.resource.data.text.size() > 0
                    && request.resource.data.text.size() < 500;

      allow update: if isModerator();
      allow delete: if isModerator();
    }

    // Zľavové kódy – rovnaký princíp moderácie ako pri dealoch.
    match /coupons/{couponId} {
      allow read: if resource.data.status == 'approved' || isModerator();

      allow create: if isAdmin();

      allow create: if (request.resource.data.status == 'pending'
                        || (request.resource.data.status == 'approved' && bezModeracie('couponsRequired')))
                    && bezCudzichId() && autoId(couponId)
                    && request.resource.data.author is string
                    && request.resource.data.author.size() > 0
                    && request.resource.data.author.size() < 40
                    && request.resource.data.code is string
                    && request.resource.data.code.size() > 0
                    && request.resource.data.code.size() < 40
                    && request.resource.data.store is string
                    && request.resource.data.discount is string
                    && request.resource.data.url is string
                    && request.resource.data.votes == 0;

      allow update: if isModerator()
                    || (request.auth != null
                        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['votes'])
                        && request.resource.data.votes >= resource.data.votes - 2
                        && request.resource.data.votes <= resource.data.votes + 2);

      allow delete: if isAdmin();

      match /voters/{voterId} {
        allow read: if isAdmin() || (request.auth != null && request.auth.uid == voterId);
        allow write: if request.auth != null && request.auth.uid == voterId;
      }
    }

    // Účty návštevníkov (registrácia e-mailom). Každý číta a mení len
    // svoj profil; admin číta všetky kvôli exportu pre newsletter.
    //  - e-mail v profile sa musí zhodovať s prihláseným účtom, takže
    //    si nikto nezapíše newsletter na cudziu adresu;
    //  - "overeny" smie byť true len vtedy, keď Firebase potvrdil
    //    e-mail - newsletter ide len na overené adresy (double opt-in);
    //  - polia sú vymenované, nič iné sa do profilu nedostane.
    function platnyProfil() {
      let d = request.resource.data;
      return d.keys().hasOnly(['email', 'meno', 'newsletter', 'temy', 'frekvencia', 'letiska',
                               'kraj', 'ulozene', 'overeny', 'suhlasPodmienky', 'suhlasNewsletter',
                               'vytvorene', 'upravene',
                               // výhody pre registrovaných: strážca dealov, letenky, história
                               'strazcovia', 'letenkyMax', 'historiaDealy', 'historiaHladania'])
        && d.email == request.auth.token.email
        && d.newsletter is bool
        && (!('meno' in d) || (d.meno is string && d.meno.size() <= 40))
        && (!('temy' in d) || (d.temy is list && d.temy.size() <= 10))
        && (!('letiska' in d) || (d.letiska is list && d.letiska.size() <= 5))
        && (!('frekvencia' in d) || d.frekvencia in ['tyzden', 'den'])
        && (!('kraj' in d) || (d.kraj is string && d.kraj.size() <= 40))
        && (!('ulozene' in d) || (d.ulozene is list && d.ulozene.size() <= 500))
        && (!('overeny' in d) || d.overeny == false || request.auth.token.email_verified == true)
        && (!('strazcovia' in d) || (d.strazcovia is list && d.strazcovia.size() <= 10))
        && (!('letenkyMax' in d) || (d.letenkyMax is number && d.letenkyMax >= 0 && d.letenkyMax <= 200))
        && (!('historiaDealy' in d) || (d.historiaDealy is list && d.historiaDealy.size() <= 40))
        && (!('historiaHladania' in d) || (d.historiaHladania is list && d.historiaHladania.size() <= 20));
    }
    match /users/{uid} {
      allow read: if isAdmin() || (request.auth != null && request.auth.uid == uid);
      allow create, update: if request.auth != null && request.auth.uid == uid
                            && request.auth.token.email is string
                            && platnyProfil();
      allow delete: if request.auth != null && request.auth.uid == uid;
    }

    // Nastavenia moderácie – verejné čítanie, zápis len admin.
    match /settings/{settingId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Návštevnosť – 1 záznam na deň a návštevníka. ID dokumentu obsahuje
    // dátum aj uid, takže si nikto neprepíše cudzí záznam.
    // Návštevy. Jeden dokument = jedno otvorenie stránky, nie jeden deň:
    // ID je "datum_uid_nahodnecislo", aby sa dala zaznamenať aj opakovaná
    // návšteva toho istého človeka a jej dĺžka.
    //
    // Dĺžku dopĺňa prehliadač až počas návštevy, preto je tu povolený
    // update - ale len vlastnému záznamu a len na dve polia. Bez toho
    // obmedzenia by si ktokoľvek mohol prepísať mesto alebo krajinu
    // v cudzom zázname a štatistiky by boli na nič.
    match /visits/{visitId} {
      allow read: if isAdmin();
      allow create: if request.auth != null
                    && visitId.split('_')[1] == request.auth.uid
                    && request.resource.data.uid == request.auth.uid
                    && request.resource.data.country is string
                    && request.resource.data.date is string;
      // Okrem dĺžky pribúda počas návštevy aj počet preklikov na dealy.
      // Drží sa tu zámerne: kolekcia deal_clicks vie, KTORÝ deal bol
      // kliknutý, ale nie z akej návštevy - a bez toho sa nedá povedať,
      // ktorý zdroj návštevnosti prináša ľudí, čo naozaj klikajú.
      allow update: if request.auth != null
                    && resource.data.uid == request.auth.uid
                    && request.resource.data.diff(resource.data)
                         .affectedKeys().hasOnly(['duration', 'lastSeen', 'clicks'])
                    && (!('duration' in request.resource.data.diff(resource.data).affectedKeys())
                        || (request.resource.data.duration is int
                            && request.resource.data.duration >= 0));
      allow delete: if false;
    }

    // ═══════════════════════════════════════════════════════════════
    //  NOVÉ KOLEKCIE
    // ═══════════════════════════════════════════════════════════════

    // Predajcovia a ich odkazy – spravuje ich admin v admin paneli.
    // Agent ich číta cez service account, ktorý pravidlá obchádza,
    // takže verejné čítanie netreba.
    match /merchants/{merchantId} {
      allow read, write: if isAdmin();
    }

    // Záznam zmien stavu dealov.
    // Úprava a mazanie sú zakázané úplne – denník, ktorý sa dá
    // prepísať, nie je denník.
    match /audit_log/{entryId} {
      allow read:   if isModerator();
      // Záznam musí niesť e-mail toho, kto ho zapisuje - moderátor sa
      // nevydá za majiteľa.
      allow create: if isModerator()
                    && (!('by' in request.resource.data) || request.resource.data.by == request.auth.token.email);
      allow update, delete: if false;
    }

    // Prekliky na dealy – jediná kolekcia, kam smie zapisovať bežný
    // návštevník. Inak by sa prekliky nedali merať.
    //
    // Obmedzenie na dve polia tu je, ale presná kontrola "presne +1"
    // zámerne NIE: klient používa increment() a nepodarilo sa mi
    // spoľahlivo doložiť, či sú takéto serverové transformácie v
    // pravidlách viditeľné. Keby neboli, prísnejšia podmienka by
    // počítadlo ticho zablokovala a ty by si to nespozoroval.
    //
    // Znamená to, že odhodlaný človek si vie číslo nafúknuť. Je to
    // štatistika len pre teba a nič sa podľa nej neradí, takže je to
    // prijateľná cena. Keby na presnosti záležalo, počítanie sa
    // presunie do Cloud Function.
    match /deal_clicks/{dealId} {
      allow read: if isAdmin();

      allow create: if request.auth != null
                    && request.resource.data.keys().hasOnly(['count', 'lastClick']);

      allow update: if request.auth != null
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['count', 'lastClick']);

      allow delete: if isAdmin();
    }

    // Ceny základných potravín z národného porovnávača.
    // Číta ich ktokoľvek (je to obsah stránky), zapisuje len agent
    // cez service account.
    match /food_prices/{docId} {
      allow read: if true;
      allow write: if false;
    }

    // ── Admin zóna: role, úlohy, financie, marketing ─────────────────
    // Ľudia s prístupom do adminu. Každý si smie prečítať svoj záznam
    // (admin podľa neho vie, čo mu ukázať); meniť ich smie len majiteľ.
    match /admini/{email} {
      allow read: if isAdmin() || (overenyUcet() && request.auth.token.email == email);
      allow write: if jeMajitel();
    }

    // Úlohy pre plánovač (načítanie odkazu, kontrola platnosti,
    // spustenie workflow). Admin ich len zadáva; vykonáva ich a výsledok
    // zapisuje plánovač cez service account.
    match /admin_ulohy/{id} {
      allow read: if isAdmin();
      allow create: if isAdmin()
                    && request.resource.data.stav == 'caka'
                    && request.resource.data.typ in ['nacitaj_url', 'kontrola_platnosti', 'spusti_workflow', 'zopakuj_beh', 'test_email'];
      allow update, delete: if false;
    }

    // Údaje len pre admin: info od agenta, provízie, príjmy, príspevky
    // na Facebook, UTM odkazy, pripomienky.
    match /admin_info/{id} { allow read: if isAdmin(); allow write: if false; }
    match /provizie/{id} { allow read, write: if isAdmin(); }
    match /financie/{id} { allow read, write: if isAdmin(); }
    match /fb_posty/{id} { allow read, write: if isAdmin(); }
    match /utm_odkazy/{id} { allow read, write: if isAdmin(); }
    // Nastavenia e-mailov (odosielateľ, server, šablóny). Heslo tu nie je -
    // je len v tajných nastaveniach GitHubu.
    match /nastavenia_admin/{id} { allow read, write: if isAdmin(); }
    // Odoslané e-maily zapisuje plánovač; admin ich číta a zmazaním
    // záznamu povolí odoslanie znova.
    match /email_log/{id} { allow read, delete: if isAdmin(); allow create, update: if false; }
    match /pripomienky/{id} {
      allow read, update, delete: if isAdmin();
      allow create: if isAdmin() && request.resource.data.odoslane == false;
    }

    // ── Merania z verejnej stránky ───────────────────────────────────
    // Čo ľudia hľadajú. Len text hľadania, sekcia a počet výsledkov -
    // nič, čo by identifikovalo človeka.
    match /hladania/{id} {
      allow read, delete: if isAdmin();
      allow create: if request.auth != null && autoId(id)
                    && request.resource.data.keys().hasOnly(['q', 'sekcia', 'pocet', 'date', 'ts'])
                    && request.resource.data.q is string
                    && request.resource.data.q.size() > 0 && request.resource.data.q.size() <= 80
                    && request.resource.data.sekcia in ['dealy', 'kody', 'potraviny']
                    && request.resource.data.pocet is int
                    && request.resource.data.date is string
                    && request.resource.data.date.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$');
    }
    // Počítadlá na deň: používanie funkcií a prekliky po dňoch. ID
    // dokumentu je dátum, polia sú počty. Ako pri deal_clicks sa presné
    // "+1" overiť nedá - je to štatistika len pre admin.
    match /pouzitie/{den} {
      allow read: if isAdmin();
      allow create, update: if request.auth != null && den.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')
                            && request.resource.data.size() <= 300;
    }
    match /kliky_den/{den} {
      allow read: if isAdmin();
      allow create, update: if request.auth != null && den.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')
                            && request.resource.data.size() <= 3000;
    }

    // Žiadosť o e-mail o účte (overenie adresy, nové heslo). Píše ju
    // stránka, e-mail potom pošle plánovač v našom vzhľade - Firebase má
    // úpravu svojich šablón pre tento projekt zakázanú.
    //
    // Overenie si smie vypýtať len prihlásený a len na svoju vlastnú
    // adresu. Obnovu hesla musí vedieť vyvolať aj neprihlásený (práve
    // sa dnu nedostane), preto tu stačí ľubovoľná adresa - plánovač
    // posiela najviac päť takých e-mailov na adresu za hodinu a na
    // neexistujúci účet nepošle nič.
    match /ucty_akcie/{id} {
      allow read: if isAdmin();
      allow create: if request.auth != null && autoId(id)
                    && request.resource.data.stav == 'caka'
                    && (
                      (request.resource.data.typ == 'overenie'
                       && request.resource.data.keys().hasOnly(['typ', 'uid', 'email', 'meno', 'stav', 'vytvorene'])
                       && request.resource.data.uid == request.auth.uid
                       && request.resource.data.email == request.auth.token.email)
                      ||
                      (request.resource.data.typ == 'heslo'
                       && request.resource.data.keys().hasOnly(['typ', 'email', 'stav', 'vytvorene'])
                       && request.resource.data.email is string
                       && request.resource.data.email.size() > 4
                       && request.resource.data.email.size() <= 120)
                    );
      allow update, delete: if false;
    }

    // ── Komunita: body, odznaky, rebríček, dôvera ────────────────────
    // Počíta ich plánovač cez service account. Stránka ich len číta -
    // keby si body zapisoval prehliadač, vedel by si si ich dopísať sám.
    match /prispevatelia/{uid} {
      allow read: if true;
      allow write: if false;
    }
    match /verejne/{id} {
      allow read: if true;
      allow write: if false;
    }
    // Kto smie zverejňovať dealy bez čakania. Každý vidí len svoj záznam
    // (stránka podľa neho vie, čo ti má napísať), admin všetky.
    match /duveryhodni/{uid} {
      allow read: if isAdmin() || (request.auth != null && request.auth.uid == uid);
      allow write: if isAdmin();
    }

    // Kolekcie agent_state a price_watch tu zámerne nie sú.
    // Zapisuje do nich výhradne agent cez service account, ktorý
    // pravidlá obchádza. Čokoľvek bez vlastného pravidla je pre
    // klientov automaticky zakázané – presne to chceme.
  }
}
